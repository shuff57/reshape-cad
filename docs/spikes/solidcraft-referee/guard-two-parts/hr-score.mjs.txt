  const wall = Math.round(performance.now() - t0);
  if (p.signal === 'SIGKILL' || p.error?.code === 'ETIMEDOUT') return { ok: false, timedOut: true, cpuMs: wall, failed: { command: '(budget)', error: `TIME BUDGET: exceeded ${Math.round(s * 1000)} ms` } };
  if (p.status !== 0 || p.error) console.error('PROBE FAIL', p.status, p.signal, p.error?.code, String(p.stderr).slice(0, 300));
  if (p.status !== 0) return { ok: false, probeError: true, failed: { command: '(probe)', error: `probe exit ${p.status} ${p.signal} ${String(p.stderr).slice(0, 200)}` } };
  let o; try { o = JSON.parse(String(p.stdout)); } catch (e) { console.error('PARSE FAIL', String(p.stdout).length, String(p.stdout).slice(0, 200), String(p.stderr).slice(0, 300)); throw e; } if (!o.res.ok) console.error('PROBE NOT OK', s, JSON.stringify(o.res.failed ?? o.res).slice(0, 300)); return { ...o.res, cpuMs: o.ms };

// score a built hollow: oracle P (plane-offset, h 0.05) on volume and (rect models) centre of mass, and the Euclidean-erosion cross-check where it is valid
// (rect models with holes only, no cuts); the model frame is z0 (box z 0..H) or centred (z -H/2..H/2: the oracle COM is shifted by -H/2)
function score(c, scene) {
  if (!c.model) return { noOracle: true };
  const o = oracle(c.model, { h: 0.05 });
  const bb = c.model.mode === 'convex' ? null : c.model.boxes[0];
  const Hh = c.model.mode === 'convex' ? c.model.convex.H : bb[5] - bb[4];
  const dz = c.frame === 'centred' ? -Hh / 2 : 0;
  const diag = c.model.mode === 'convex' ? Math.hypot(c.model.convex.W, c.model.convex.D, Hh) : Math.hypot(bb[1] - bb[0], bb[3] - bb[2], Hh);
  const out = { P: o.P, dVol: (scene.volume - o.P) / o.P };
  if (o.com) { const oc = [o.com[0], o.com[1], o.com[2] + dz]; out.oracleCom = oc; out.dComOfDiag = Math.max(...scene.com.map((v, i) => Math.abs(v - oc[i]))) / diag; }
  if (!NO_EUCLID && c.model.mode === 'rect' && !(c.model.cuts ?? []).length && (c.model.holes ?? []).length && !(c.model.rbosses ?? []).length && c.model.boxes.length === 1) { try { const e = euclid(c.model, { h: 0.05 }); out.euclidP = e.P; out.dVolEuclid = (scene.volume - e.P) / e.P; } catch (e) { out.euclidErr = String(e.message); } }
  out.verdict = Math.abs(out.dVol) <= 1e-3 && (out.dComOfDiag === undefined || out.dComOfDiag <= 1e-3) ? 'RIGHT' : 'WRONG';
  return out;
}
